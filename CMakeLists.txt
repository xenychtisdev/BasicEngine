#[[

Build Instructions for Standard CMake:
    * Open a terminal in the root folder and run:
        . cmake -S . -B build
        . cmake --build build
    * If build already contains a CMake configuration
        generated by a different toolchain, CMake may
        report a generator mismatch. In that case, use
        a fresh directory such as build-msvc instead of
        deleting files blindly.

Setup Instructions for VSCode CMake with CMake Tools Extention.
    1. Open the Command Palette with Ctrl+Shift+P.
    2. Search for CMake: Select a Kit and select it.
    3. Choose an installed Visual Studio Build Tools kit,
        preferably an x64 MSVC compiler.
    4. Open the Command Palette again and run CMake: Configure.
    * CMake Tools will try to locate CMake and configure your
        project. If it succeeds, you'll see configuration output
        in the VS Code terminal or Output panel.
    * If no compiler kit appears, check that your Visual Studio
        Build Tools installation includes the Desktop development
        with C++ workload and a Windows SDK.

Build and Run the Program (Once CMake is set up):
    1. Configure: Command Palette -> CMake: Configure
    2. Build: Click Build in the status bar, or use
        CMake: Build from the Command Palette.
    3. Run: Choose CMake: Run Without Debugging from
        the Command Palette, or use CMake: Debug to
        launch under the debugger.

]]



cmake_minimum_required(VERSION 3.20)

project(
    BasicEngine
    VERSION 0.1.0
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(FetchContent)

# Download GLFW automatically
FetchContent_Declare(
    glfw
    GIT_REPOSITORY https://github.com/glfw/glfw.git
    GIT_TAG 3.5.1
)

# Disable GLFW features we don't need yet
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(glfw)

#add_subdirectory(include/glfw-3.5.1)

# Build BasicEngine
add_executable(
    BasicEngine
    src/main.cpp
)

# Include directories
target_include_directories(
    BasicEngine
    PRIVATE
       ${CMAKE_CURRENT_SOURCE_DIR}/include
)

# Link relevant directories
target_link_libraries(
    BasicEngine
    PRIVATE
        glfw
)
